cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(strata_distributed_runtime LANGUAGES CXX)
  enable_testing()
endif()
get_filename_component(_tp_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(Threads REQUIRED)
add_library(strata_tp_runtime STATIC mpi_runtime.cpp)
target_compile_features(strata_tp_runtime PUBLIC cxx_std_20)
target_include_directories(strata_tp_runtime PUBLIC "${_tp_root}/include")
target_link_libraries(strata_tp_runtime PUBLIC MPI::MPI_CXX Threads::Threads)
add_executable(strata-tp-mpi-runtime-test "${_tp_root}/tests/tp/mpi_runtime_test.cpp")
target_link_libraries(strata-tp-mpi-runtime-test PRIVATE strata_tp_runtime)
foreach(_ranks 1 2 3 4)
  add_test(NAME tp_mpi_${_ranks}_ranks COMMAND "${MPIEXEC_EXECUTABLE}" ${MPIEXEC_NUMPROC_FLAG} ${_ranks}
    ${MPIEXEC_PREFLAGS} $<TARGET_FILE:strata-tp-mpi-runtime-test> ${MPIEXEC_POSTFLAGS})
  set_tests_properties(tp_mpi_${_ranks}_ranks PROPERTIES TIMEOUT 15)
endforeach()
foreach(_mode mismatch long inplace)
  add_test(NAME tp_mpi_${_mode} COMMAND "${MPIEXEC_EXECUTABLE}" ${MPIEXEC_NUMPROC_FLAG} 2
    ${MPIEXEC_PREFLAGS} $<TARGET_FILE:strata-tp-mpi-runtime-test> ${MPIEXEC_POSTFLAGS} ${_mode})
  set_tests_properties(tp_mpi_${_mode} PROPERTIES TIMEOUT 15)
endforeach()
set_tests_properties(tp_mpi_mismatch PROPERTIES WILL_FAIL TRUE)
add_executable(strata-tp-packet-test "${_tp_root}/tests/tp/packet_test.cpp")
target_link_libraries(strata-tp-packet-test PRIVATE strata_tp_runtime)
add_test(NAME tp_packet COMMAND strata-tp-packet-test)
set_tests_properties(tp_packet PROPERTIES TIMEOUT 10)

option(STRATA_TP_CUDA "Build the CUDA/NCCL distributed runtime" OFF)
if(STRATA_TP_CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_path(STRATA_NCCL_INCLUDE_DIR nccl.h HINTS ENV NCCL_ROOT PATH_SUFFIXES include)
  find_library(STRATA_NCCL_LIBRARY NAMES nccl nccl.so.2 libnccl.so.2 HINTS ENV NCCL_ROOT PATH_SUFFIXES lib lib64)
  if(NOT STRATA_NCCL_INCLUDE_DIR OR NOT STRATA_NCCL_LIBRARY)
    message(FATAL_ERROR "NCCL headers and library are required; set STRATA_NCCL_INCLUDE_DIR and STRATA_NCCL_LIBRARY")
  endif()
  add_library(strata_tp_nccl STATIC nccl_session.cpp)
  target_include_directories(strata_tp_nccl PUBLIC "${STRATA_NCCL_INCLUDE_DIR}")
  target_link_libraries(strata_tp_nccl PUBLIC strata_tp_runtime CUDA::cudart "${STRATA_NCCL_LIBRARY}")
  add_executable(strata-tp-nccl-runtime-test "${_tp_root}/tests/tp/nccl_runtime_test.cpp")
  target_link_libraries(strata-tp-nccl-runtime-test PRIVATE strata_tp_nccl)
endif()
